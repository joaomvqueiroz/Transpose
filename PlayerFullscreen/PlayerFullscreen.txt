<script>
  (function() {
    const overlay = document.getElementById('controls-overlay');
    const playPauseBtn = document.getElementById('btn-play-pause');
    const playPauseIcon = document.getElementById('play-pause-icon');
    const lockBtn = document.getElementById('btn-lock-toggle');
    const lockSurfaceBtn = document.getElementById('btn-lock-surface');
    const lockedCurtain = document.getElementById('locked-curtain');
    const unlockCurtainBtn = document.getElementById('btn-unlock-curtain');
    const playerContainer = document.getElementById('player-container');
    const speedVal = document.getElementById('speed-val');
    const pitchVal = document.getElementById('pitch-val');

    let isPlaying = true;
    let hideTimeout;

    // Toggle Play/Pause UI state
    playPauseBtn.addEventListener('click', (e) => {
      e.stopPropagation();
      isPlaying = !isPlaying;
      playPauseIcon.textContent = isPlaying ? 'pause' : 'play_arrow';
      resetHideTimer();
    });

    // Auto-hide controls overlay after 3 seconds of inactivity
    function resetHideTimer() {
      if (overlay.classList.contains('opacity-0')) {
        overlay.classList.remove('opacity-0', 'pointer-events-none');
        overlay.classList.add('opacity-100');
      }
      clearTimeout(hideTimeout);
      hideTimeout = setTimeout(() => {
        overlay.classList.remove('opacity-100');
        overlay.classList.add('opacity-0', 'pointer-events-none');
      }, 3000);
    }

    // Toggle overlay visibility on screen tap (if not clicking interactive components directly)
    playerContainer.addEventListener('click', (e) => {
      if (lockedCurtain.classList.contains('hidden')) {
        if (overlay.classList.contains('opacity-0')) {
          resetHideTimer();
        } else if (e.target === playerContainer || e.target.id === 'video-background') {
          overlay.classList.remove('opacity-100');
          overlay.classList.add('opacity-0', 'pointer-events-none');
        }
      }
    });

    // Reset timer on any overlay interaction
    overlay.addEventListener('click', () => {
      resetHideTimer();
    });

    // Lock Screen Behaviors
    function enableLock() {
      clearTimeout(hideTimeout);
      overlay.classList.remove('opacity-100');
      overlay.classList.add('opacity-0', 'pointer-events-none');
      lockedCurtain.classList.remove('hidden');
    }

    function disableLock() {
      lockedCurtain.classList.add('hidden');
      resetHideTimer();
    }

    lockBtn.addEventListener('click', (e) => {
      e.stopPropagation();
      enableLock();
    });

    lockSurfaceBtn.addEventListener('click', (e) => {
      e.stopPropagation();
      enableLock();
    });

    unlockCurtainBtn.addEventListener('click', (e) => {
      e.stopPropagation();
      disableLock();
    });

    // Speed Step Adjustment demo
    const speeds = ['0.75x', '0.80x', '0.85x', '0.90x', '1.0x'];
    let currentSpeedIdx = 2;

    document.getElementById('speed-up').addEventListener('click', (e) => {
      e.stopPropagation();
      if (currentSpeedIdx < speeds.length - 1) currentSpeedIdx++;
      speedVal.textContent = speeds[currentSpeedIdx];
      resetHideTimer();
    });

    document.getElementById('speed-down').addEventListener('click', (e) => {
      e.stopPropagation();
      if (currentSpeedIdx > 0) currentSpeedIdx--;
      speedVal.textContent = speeds[currentSpeedIdx];
      resetHideTimer();
    });

    // Initial countdown on screen launch
    resetHideTimer();
  })();
</script></main></body></html>
