<script>
    // Pitch / Key Stepper
    const chromaticScale = ['C', 'D♭', 'D', 'E♭', 'E', 'F', 'G♭', 'G', 'A♭', 'A', 'B♭', 'B'];
    let currentPitchOffset = -2; // Default -2 semitones -> Bb
    const baseKeyIndex = 0; // C

    function updatePitchUI() {
      const idx = (baseKeyIndex + currentPitchOffset + 120) % 12;
      document.getElementById('display-key').innerText = chromaticScale[idx];
      const semitoneLabel = currentPitchOffset === 0 
        ? 'Tom Original' 
        : (currentPitchOffset > 0 ? `+${currentPitchOffset} semitons` : `${currentPitchOffset} semitons`);
      document.getElementById('display-semitones').innerText = semitoneLabel;
    }

    document.getElementById('btn-pitch-down').addEventListener('click', () => {
      currentPitchOffset--;
      updatePitchUI();
    });

    document.getElementById('btn-pitch-up').addEventListener('click', () => {
      currentPitchOffset++;
      updatePitchUI();
    });

    // Speed Slider & Preset Chips
    const speedSlider = document.getElementById('speed-slider');
    const speedIndicator = document.getElementById('speed-indicator');

    function setSpeed(val) {
      const formatted = parseFloat(val).toFixed(2);
      speedSlider.value = formatted;
      speedIndicator.innerText = `${formatted}x`;

      document.querySelectorAll('.speed-chip').forEach(btn => {
        if (btn.innerText.trim() === `${formatted}x`) {
          btn.classList.add('border-primary-cyan/40', 'text-primary-cyan', 'font-semibold');
          btn.classList.remove('border-card-border', 'text-text-muted');
        } else {
          btn.classList.remove('border-primary-cyan/40', 'text-primary-cyan', 'font-semibold');
          btn.classList.add('border-card-border', 'text-text-muted');
        }
      });
    }

    speedSlider.addEventListener('input', (e) => {
      setSpeed(e.target.value);
    });

    // Main Play/Pause state synchronization
    let isPlaying = true;
    const playBtn = document.getElementById('btn-play-pause');
    const playIcon = document.getElementById('play-icon');
    const fsPlayIcon = document.getElementById('fs-play-icon');

    function togglePlayState() {
      isPlaying = !isPlaying;
      const iconName = isPlaying ? 'pause' : 'play_arrow';
      playIcon.innerText = iconName;
      if (fsPlayIcon) fsPlayIcon.innerText = iconName;
    }

    playBtn.addEventListener('click', togglePlayState);

    // Context Dropdown Menu
    const moreBtn = document.getElementById('btn-more-options');
    const dropdown = document.getElementById('dropdown-menu');

    moreBtn.addEventListener('click', (e) => {
      e.stopPropagation();
      dropdown.classList.toggle('hidden');
    });

    function closeDropdown() {
      dropdown.classList.add('hidden');
    }

    document.addEventListener('click', () => {
      closeDropdown();
    });

    // Sidebar Drawer Controls
    const sidebarOverlay = document.getElementById('sidebar-overlay');
    const sidebarContent = document.getElementById('sidebar-content');
    const openSidebarBtn = document.getElementById('btn-sidebar-open');
    const closeSidebarBtn = document.getElementById('btn-sidebar-close');

    function openSidebar() {
      sidebarOverlay.classList.remove('hidden');
      setTimeout(() => {
        sidebarContent.classList.remove('-translate-x-full');
      }, 10);
    }

    function closeSidebar() {
      sidebarContent.classList.add('-translate-x-full');
      setTimeout(() => {
        sidebarOverlay.classList.add('hidden');
      }, 300);
    }

    openSidebarBtn.addEventListener('click', openSidebar);
    closeSidebarBtn.addEventListener('click', closeSidebar);
    sidebarOverlay.addEventListener('click', (e) => {
      if (e.target === sidebarOverlay) closeSidebar();
    });

    // Study Tools Accordion (Ferramentas de Estudo)
    const toggleStudyBtn = document.getElementById('btn-toggle-study');
    const studyPanel = document.getElementById('study-tools-panel');
    const studyChevron = document.getElementById('study-chevron');

    toggleStudyBtn.addEventListener('click', () => {
      const isHidden = studyPanel.classList.contains('hidden');
      if (isHidden) {
        studyPanel.classList.remove('hidden');
        studyChevron.classList.add('rotate-180');
      } else {
        studyPanel.classList.add('hidden');
        studyChevron.classList.remove('rotate-180');
      }
    });

    // Audio Advanced Drawer
    const advancedDrawer = document.getElementById('drawer-advanced');
    function openDrawer() {
      advancedDrawer.classList.remove('hidden');
      advancedDrawer.classList.add('flex');
    }
    function closeDrawer() {
      advancedDrawer.classList.add('hidden');
      advancedDrawer.classList.remove('flex');
    }

    document.getElementById('btn-open-advanced').addEventListener('click', openDrawer);
    document.getElementById('btn-close-advanced').addEventListener('click', closeDrawer);
    document.getElementById('btn-done-advanced').addEventListener('click', closeDrawer);

    // Fullscreen Overlay & Lock Feature
    const fsOverlay = document.getElementById('fullscreen-overlay');
    let isControlsLocked = false;
    const lockIcon = document.getElementById('lock-icon');
    const lockBanner = document.getElementById('fs-locked-banner');
    const fsPlaybackControls = document.getElementById('fs-playback-controls');
    const fsBottomBar = document.getElementById('fs-bottom-bar');

    function openFullscreen() {
      fsOverlay.classList.remove('hidden');
      fsOverlay.classList.add('flex');
      isControlsLocked = false;
      applyLockState();
    }

    function closeFullscreen() {
      fsOverlay.classList.add('hidden');
      fsOverlay.classList.remove('flex');
    }

    function toggleScreenLock() {
      isControlsLocked = !isControlsLocked;
      applyLockState();
    }

    function applyLockState() {
      if (isControlsLocked) {
        lockIcon.innerText = 'lock';
        lockIcon.classList.add('text-primary-cyan');
        lockBanner.classList.remove('hidden');
        fsPlaybackControls.classList.add('opacity-0', 'pointer-events-none');
        fsBottomBar.classList.add('opacity-0', 'pointer-events-none');
        setTimeout(() => {
          lockBanner.classList.add('hidden');
        }, 2500);
      } else {
        lockIcon.innerText = 'lock_open';
        lockIcon.classList.remove('text-primary-cyan');
        lockBanner.classList.add('hidden');
        fsPlaybackControls.classList.remove('opacity-0', 'pointer-events-none');
        fsBottomBar.classList.remove('opacity-0', 'pointer-events-none');
      }
    }
  </script>
